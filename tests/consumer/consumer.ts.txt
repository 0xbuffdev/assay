// The ES2022-only consumer needs neither Node nor DOM ambient types.
const fixtureSignal: ObservationAbortSignal = {
  aborted: false,
  addEventListener(_type, _listener, _options) {},
  removeEventListener(_type, _listener) {},
};
async function verifyCollection(): Promise<void> {
  const collected = await collectObservations(definition, evidence, port, { signal: fixtureSignal });
  if (collected.ok) {
    const claims: string | undefined = collected.receipt?.provenance.adapterId;
    void claims;
  } else if (collected.outcome === "evaluation_error") {
    const reason: "OBSERVATION_PORT_ERROR" | "OBSERVATION_CANCELLED" = collected.reasonCode;
    void reason;
  } else {
    const details: readonly { readonly path: string }[] = collected.diagnostics;
    void details;
  }
}
