    revisionDigest: definition.revisionDigest,

const port: ObservationPort = {
  async observe(request: ObservationPortRequest): Promise<unknown> {
    const ids: readonly string[] = request.criteria.map((criterion) => criterion.criterionId);
    const max: number = request.limits.maxCapturedBytes;
    void ids;
    void max;
    return {
      schemaVersion: ASSAY_SCHEMA_VERSION,
      definition: request.definition,
      observations: [],
      provenance: {
        adapterId: "fixture",
        adapterVersion: "v1",
        runRef: "run1",
        toolchainId: "none",
        toolchainVersion: "v1",
        capturedStdoutBytes: 0,
        capturedStderrBytes: 0,
        stdoutTruncated: false,
        stderrTruncated: false,
      },
    };
  },
};

async function verifyCollection(): Promise<void> {
  const collected = await collectObservations(definition, evidence, port);
  if (collected.ok) {
    const claims: string | undefined = collected.receipt?.provenance.adapterId;
    void claims;
  } else if (collected.outcome === "evaluation_error") {
    const reason: "OBSERVATION_PORT_ERROR" = collected.reasonCode;
    void reason;
  } else {
    const details: readonly { readonly path: string }[] = collected.diagnostics;
    void details;
  }
}
